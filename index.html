        // 音效系統 (使用 Web Audio API 生成，免外部資源)

        const AudioContext = window.AudioContext || window.webkitAudioContext;

        let audioCtx;


        function initAudio() {

            if (!audioCtx) {

                audioCtx = new AudioContext();

            }

            if (audioCtx.state === 'suspended') {

                audioCtx.resume();

            }

        }


        function playTone(freq, type, duration, vol=0.1) {

            if(!audioCtx) return;

            const osc = audioCtx.createOscillator();

            const gain = audioCtx.createGain();

            osc.type = type;

            osc.frequency.setValueAtTime(freq, audioCtx.currentTime);

            gain.gain.setValueAtTime(vol, audioCtx.currentTime);

            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);

            osc.connect(gain);

            gain.connect(audioCtx.destination);

            osc.start();

            osc.stop(audioCtx.currentTime + duration);

        }


        function playShootSound() {

            if(!audioCtx) return;

            const osc = audioCtx.createOscillator();

            const gain = audioCtx.createGain();

            osc.type = 'square';

            osc.frequency.setValueAtTime(800, audioCtx.currentTime);

            osc.frequency.exponentialRampToValueAtTime(100, audioCtx.currentTime + 0.1);

            gain.gain.setValueAtTime(0.05, audioCtx.currentTime);

            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);

            osc.connect(gain);

        function createGridTexture() {

            const canvas = document.createElement('canvas');

            canvas.width = 256; canvas.height = 256;

            const ctx = canvas.getContext('2d');

            ctx.fillStyle = '#1a1a1a'; ctx.fillRect(0, 0, 256, 256);

            ctx.strokeStyle = '#333333'; ctx.lineWidth = 4; ctx.strokeRect(0, 0, 256, 256);

            const texture = new THREE.CanvasTexture(canvas);

            texture.wrapS = THREE.RepeatWrapping; texture.wrapT = THREE.RepeatWrapping;

            texture.repeat.set(10, 10);

            return texture;

        }

        // 建立 SDG 牌子紋理 (有 Emoji 圖示)

        // -----------------------------------------------------

        function createSDGTexture(sdgObj) {

            const canvas = document.createElement('canvas');

            canvas.width = 512;

            canvas.height = 512;

            const ctx = canvas.getContext('2d');

            const text = sdgObj.text;

            const icon = sdgObj.icon;

            

            // 背景與邊框

            ctx.fillStyle = '#ffffff';

            ctx.fillRect(0, 0, 512, 512);

            ctx.strokeStyle = '#e74c3c';

            ctx.lineWidth = 20;

            ctx.strokeRect(0, 0, 512, 512);

            

            // 繪製 Emoji 圖示

            ctx.textAlign = 'center';

            ctx.textBaseline = 'middle';

            ctx.font = '100px sans-serif'; // emoji 大小

            ctx.fillText(icon, 256, 110);

            

            const splitIndex = text.indexOf(': ');

            if(splitIndex !== -1) {

                const title = text.substring(0, splitIndex);

                const desc = text.substring(splitIndex + 2);

                

                ctx.fillStyle = '#c0392b';

                ctx.font = 'bold 45px sans-serif';

                ctx.fillText(title, 256, 220);

                

                ctx.fillStyle = '#2c3e50';

                ctx.font = 'bold 36px sans-serif';

                const words = desc.split(' ');

                let line = '';

                let y = 300;

                for(let n = 0; n < words.length; n++) {

                    let testLine = line + words[n] + ' ';

                    let metrics = ctx.measureText(testLine);

                    if (metrics.width > 420 && n > 0) {

                        ctx.fillText(line, 256, y);

                        line = words[n] + ' ';

                        y += 50;

                    } else {

                        line = testLine;

                    }

                }

                ctx.fillText(line, 256, y);

            } else {

                ctx.fillStyle = '#2c3e50';

                ctx.font = 'bold 45px sans-serif';

                ctx.fillText(text, 256, 300);

            }

            

            return new THREE.CanvasTexture(canvas);

        }

        // 控制器 (支援雙平台)

        // -----------------------------------------------------

        function setupControls() {

            const moveZone = document.getElementById('move-zone');

            const joystick = document.getElementById('joystick');

            const lookZone = document.getElementById('look-zone');

            const shootBtn = document.getElementById('shoot-btn');


            let moveTouchId = null;

            let lookTouchId = null;

            let moveStartX = 0;

            let moveStartY = 0;


            moveZone.addEventListener('touchstart', (e) => {

                if (isPCMode) return;

                e.preventDefault();

                if (moveTouchId !== null) return;

                const touch = e.changedTouches[0];

                moveTouchId = touch.identifier;

                const rect = moveZone.getBoundingClientRect();

                moveStartX = rect.left + rect.width / 2;

                moveStartY = rect.top + rect.height / 2;

                updateJoystick(touch.clientX, touch.clientY);

            });


            moveZone.addEventListener('touchmove', (e) => {

                if (isPCMode) return;

                e.preventDefault();

                for (let i=0; i<e.changedTouches.length; i++) {

                    if (e.changedTouches[i].identifier === moveTouchId) {

                        updateJoystick(e.changedTouches[i].clientX, e.changedTouches[i].clientY);

                    }

                }

            });


            const endMove = (e) => {

                if (isPCMode) return;

                e.preventDefault();

                for (let i=0; i<e.changedTouches.length; i++) {

                    if (e.changedTouches[i].identifier === moveTouchId) {

                        moveTouchId = null;

                        joystick.style.transform = `translate(-50%, -50%)`;

                        moveForward = 0;

                        moveRight = 0;

                    }

                }

            };

            moveZone.addEventListener('touchend', endMove);

            moveZone.addEventListener('touchcancel', endMove);


            function updateJoystick(x, y) {

                const dx = x - moveStartX; const dy = y - moveStartY;

                const maxRadius = 50;

                let distance = Math.sqrt(dx*dx + dy*dy);

                let angle = Math.atan2(dy, dx);

                if (distance > maxRadius) distance = maxRadius;

                const handleX = distance * Math.cos(angle);

                const handleY = distance * Math.sin(angle);

                joystick.style.transform = `translate(calc(-50% + ${handleX}px), calc(-50% + ${handleY}px))`;

                moveRight = handleX / maxRadius;

                moveForward = handleY / maxRadius;

            }


            let lastLookX = 0; let lastLookY = 0;

            lookZone.addEventListener('touchstart', (e) => {

                if (isPCMode) return;

                e.preventDefault();

                if (lookTouchId !== null) return;

                const touch = e.changedTouches[0];

                lookTouchId = touch.identifier;

                lastLookX = touch.clientX; lastLookY = touch.clientY;

            });


            lookZone.addEventListener('touchmove', (e) => {

                if (isPCMode) return;

                e.preventDefault();

                for (let i=0; i<e.changedTouches.length; i++) {

                    if (e.changedTouches[i].identifier === lookTouchId && !isModalOpen) {

                        const deltaX = e.changedTouches[i].clientX - lastLookX;

                        const deltaY = e.changedTouches[i].clientY - lastLookY;

                        yawObject.rotation.y -= deltaX * lookSensitivity;

                        pitchObject.rotation.x -= deltaY * lookSensitivity;

                        pitchObject.rotation.x = Math.max(-Math.PI/2, Math.min(Math.PI/2, pitchObject.rotation.x));

                        lastLookX = e.changedTouches[i].clientX;

                        lastLookY = e.changedTouches[i].clientY;

                    }

                }

            });


            const endLook = (e) => {

                if (isPCMode) return;

                for (let i=0; i<e.changedTouches.length; i++) {

                    if (e.changedTouches[i].identifier === lookTouchId) lookTouchId = null;

                }

            };

            lookZone.addEventListener('touchend', endLook);

            lookZone.addEventListener('touchcancel', endLook);


            shootBtn.addEventListener('touchstart', (e) => {

                if (isPCMode) return;

                e.preventDefault();

                if (!isModalOpen) shoot();

            });


            // 電腦版鍵盤事件

            window.addEventListener('keydown', (e) => {

                if (isModalOpen || !isPCMode) return;

                const k = e.key;

                if (k === 'w' || k === 'W') keys.w = true;

                if (k === 'a' || k === 'A') keys.a = true;

                if (k === 's' || k === 'S') keys.s = true;

                if (k === 'd' || k === 'D') keys.d = true;

                if (k === 'ArrowUp') keys.ArrowUp = true;

                if (k === 'ArrowDown') keys.ArrowDown = true;

                if (k === 'ArrowLeft') keys.ArrowLeft = true;

                if (k === 'ArrowRight') keys.ArrowRight = true;

                

                // 空白鍵統一觸發射擊(或開門)

                if (e.code === 'Space' && !isModalOpen) {

                    e.preventDefault();

                    shoot();

                }

            });


            window.addEventListener('keyup', (e) => {

                if (!isPCMode) return;

                const k = e.key;

                if (k === 'w' || k === 'W') keys.w = false;

                if (k === 'a' || k === 'A') keys.a = false;

                if (k === 's' || k === 'S') keys.s = false;

                if (k === 'd' || k === 'D') keys.d = false;

                if (k === 'ArrowUp') keys.ArrowUp = false;

                if (k === 'ArrowDown') keys.ArrowDown = false;

                if (k === 'ArrowLeft') keys.ArrowLeft = false;

                if (k === 'ArrowRight') keys.ArrowRight = false;

            });

        }

        // 準星射擊 (敵人 / 開門 雙功能)

        function shoot() {

            initAudio(); // 確保音效已初始化

            playShootSound(); // 播放射擊音效


            const raycaster = new THREE.Raycaster();

            raycaster.setFromCamera(new THREE.Vector2(0, 0), camera);


            if (!isQuestionPhase) {

                // 射擊敵人

                const intersects = raycaster.intersectObjects(enemies.map(e => e.mesh));

                if (intersects.length > 0) {

                    const hitMesh = intersects[0].object;

                    const index = enemies.findIndex(e => e.mesh === hitMesh);

                    

                    if (index !== -1) {

                        scene.remove(hitMesh);

                        enemies.splice(index, 1);

                        score += 10;

                        updateHUD();


                        if (enemies.length === 0) {

                            spawnDoors();

                        }

                    }

                }

            } else {

                // 瞄準射擊開門

                const intersects = raycaster.intersectObjects(doors.map(d => d.mesh));

                if (intersects.length > 0) {

                    const hitMesh = intersects[0].object;

                    const doorData = doors.find(d => d.mesh === hitMesh);

                    

                    if (doorData.isCorrect) {

                        // 答對了

                        playDoorOpenSound(); // 播放開門音效

                        score += 100;

                        currentRoom++;

                        if (currentRoom > totalRooms) {

                            endGame();

                        } else {

                            loadRoom();

                        }

                    } else {

                        // 答錯處理

                        playWrongSound(); // 播放錯誤音效

                        wrongAttempts++;

                        score -= 50;

                        updateHUD();

                        

                        if (wrongAttempts >= 2) {

                            // 錯了2次，顯示解析

                            showExplanationModal();

                        } else {

                            showModal('wrong-modal');

                        }

                    }

        function animate() {

            requestAnimationFrame(animate);


            const time = performance.now();

            const delta = (time - prevTime) / 1000;

            prevTime = time;


            if (!isModalOpen) {

                if (isPCMode) {

                    const rotSpeed = 1.8 * delta;

                    if (keys.ArrowLeft) yawObject.rotation.y += rotSpeed;

                    if (keys.ArrowRight) yawObject.rotation.y -= rotSpeed;

                    if (keys.ArrowUp) pitchObject.rotation.x += rotSpeed;

                    if (keys.ArrowDown) pitchObject.rotation.x -= rotSpeed;

                    pitchObject.rotation.x = Math.max(-Math.PI/2, Math.min(Math.PI/2, pitchObject.rotation.x));


                    moveForward = (keys.s ? 1 : 0) - (keys.w ? 1 : 0);

                    moveRight = (keys.d ? 1 : 0) - (keys.a ? 1 : 0);

                }


                if (isQuestionVisible && isQuestionPhase) {

                    currentQuestionTimer -= delta;

                    if (currentQuestionTimer <= 0) {

                        currentQuestionTimer = 0;

                        isQuestionVisible = false;

                        document.getElementById('ingame-question').style.opacity = '0';

                    }

                    document.getElementById('hud-time').innerText = Math.ceil(currentQuestionTimer);

                }


                const direction = new THREE.Vector3();

                const rotation = yawObject.rotation.y;

                direction.z = moveForward; 

                direction.x = moveRight;

                direction.applyAxisAngle(new THREE.Vector3(0, 1, 0), rotation);


                // 阻止玩家靠太近門，以維持準星射擊的空間

                if (isQuestionPhase && yawObject.position.z + (direction.z * speed * delta) < -5) {

                    // 不能再往前走

                } else {

                    yawObject.position.addScaledVector(direction, speed * delta);

                }

                

                yawObject.position.x = Math.max(-9, Math.min(9, yawObject.position.x));

                yawObject.position.z = Math.max(-9, Math.min(9, yawObject.position.z));


                if (!isQuestionPhase) {

                    enemies.forEach(e => {

                        const dir = new THREE.Vector3();

                        dir.subVectors(yawObject.position, e.mesh.position);

                        dir.y = 0;

                        dir.normalize();

                        e.mesh.position.addScaledVector(dir, e.speed * delta);

                        e.mesh.lookAt(yawObject.position.x, e.mesh.position.y, yawObject.position.z);

                    });

                }

            }


            renderer.render(scene, camera);

        }

        async function fetchLeaderboard() {

            showModal('leaderboard-screen');

            const listEl = document.getElementById('leaderboard-list');

            

            // 若在外部平台且無提供資料庫設定，則不讀取

            if (!window.db) {

                listEl.innerHTML = '<div class="text-center text-gray-400">目前無資料庫連線，無法讀取排行榜。</div>';

                return;

            }


            listEl.innerHTML = '<div class="text-center text-gray-400">載入中...</div>';


            try {

                const colRef = window.collection(window.db, 'artifacts', window.appId, 'public', 'data', 'leaderboard');

                const snap = await window.getDocs(colRef);

                

                let records = [];

                snap.forEach(doc => {

                    records.push(doc.data());

                });


                records.sort((a, b) => {

                    if (b.score === a.score) return a.timeSpent - b.timeSpent;

                    return b.score - a.score;

                });


                records = records.slice(0, 10);


                listEl.innerHTML = '';

                if (records.length === 0) {

                    listEl.innerHTML = '<div class="text-center text-gray-400">尚無紀錄</div>';

                } else {

                    records.forEach((r, idx) => {

                        const div = document.createElement('div');

                        div.className = 'flex justify-between border-b border-gray-700 py-1';

                        div.innerHTML = `

                            <span class="w-8">${idx + 1}.</span>

                            <span class="flex-1 font-bold text-blue-300 truncate px-2">${r.name}</span>

                            <span class="w-16 text-right text-yellow-400">${r.score}</span>

                            <span class="w-16 text-right text-gray-400">${r.timeSpent}s</span>

                        `;

                        listEl.appendChild(div);

                    });

                }

            } catch (err) {

                console.error("Fetch leaderboard error:", err);

                listEl.innerHTML = '<div class="text-center text-red-400">無法載入排行榜</div>';

            }

        }
